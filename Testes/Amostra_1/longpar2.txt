public class Employee {
    private String name;
    private String id;
    private String position;
    private double salary;
    private String department;
    private String email;
    private String phone;
    private String address;
    private String dateOfJoining;

    public Employee(String name, String id, String position, double salary, String department, String email,
                    String phone, String address, String dateOfJoining) {
        this.name = name;
        this.id = id;
        this.position = position;
        this.salary = salary;
        this.department = department;
        this.email = email;
        this.phone = phone;
        this.address = address;
        this.dateOfJoining = dateOfJoining;
    }

    // Métodos para gerenciar o empregado...
}
