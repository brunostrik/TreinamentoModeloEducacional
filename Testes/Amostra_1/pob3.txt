public class DateExample {

    public static void main(String[] args) {
        // Criando um evento com data
        Event event = new Event("Reunião", 2025, 2, 23);

        // Exibindo as informações do evento
        System.out.println("Evento: " + event.getName());
        System.out.println("Data: " + event.getYear() + "-" + event.getMonth() + "-" + event.getDay());
    }

}

class Event {
    private String name;
    private int year;
    private int month;
    private int day;

    public Event(String name, int year, int month, int day) {
        this.name = name;
        this.year = year;
        this.month = month;
        this.day = day;
    }

    public String getName() {
        return name;
    }

    public int getYear() {
        return year;
    }

    public int getMonth() {
        return month;
    }

    public int getDay() {
        return day;
    }
}
