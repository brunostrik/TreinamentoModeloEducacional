public class TemporaryFieldExample2 {
    private int temporaryCounter; // Temporary field

    public void processItems(int[] items) {
        temporaryCounter = 0; // Initialize temporary field
        for (int item : items) {
            if (item > 10) {
                temporaryCounter++; // Temporary use of the field
            }
        }
        System.out.println("Items greater than 10: " + temporaryCounter);
    }

    public static void main(String[] args) {
        TemporaryFieldExample2 example = new TemporaryFieldExample2();
        example.processItems(new int[]{5, 15, 25, 10, 20}); // Output: Items greater than 10: 3
    }
}
