class Employee {
    private String name;
    private int id;
    private double salary;

    public Employee(String name, int id, double salary) {
        this.name = name;
        this.id = id;
        this.salary = salary;
    }

    public String getName() {
        return name;
    }

    public int getId() {
        return id;
    }

    public double getSalary() {
        return salary;
    }

    public void work() {
        System.out.println(name + " is working.");
    }

    // Método que algumas subclasses podem não precisar
    public void attendMeeting() {
        System.out.println(name + " is attending a meeting.");
    }
}

class Developer extends Employee {
    private String programmingLanguage;

    public Developer(String name, int id, double salary, String programmingLanguage) {
        super(name, id, salary);
        this.programmingLanguage = programmingLanguage;
    }

    public String getProgrammingLanguage() {
        return programmingLanguage;
    }

    @Override
    public void work() {
        System.out.println(getName() + " is coding in " + programmingLanguage + ".");
    }

    // Developer não precisa do método attendMeeting, mas o herda de Employee
}

class Salesperson extends Employee {
    private double salesTarget;

    public Salesperson(String name, int id, double salary, double salesTarget) {
        super(name, id, salary);
        this.salesTarget = salesTarget;
    }

    public double getSalesTarget() {
        return salesTarget;
    }

    @Override
    public void work() {
        System.out.println(getName() + " is achieving sales targets.");
    }

    // Salesperson não precisa do método attendMeeting, mas o herda de Employee
}

public class Main {
    public static void main(String[] args) {
        Employee dev = new Developer("Alice", 101, 90000, "Java");
        Employee sales = new Salesperson("Bob", 102, 75000, 50000);

        dev.work();
        dev.attendMeeting(); 

        sales.work();
        sales.attendMeeting();
    }
}
