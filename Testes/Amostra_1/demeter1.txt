class Engine {
    public void start() {
        System.out.println("Engine started.");
    }
}

class Car {
    private Engine engine;

    public Car() {
        engine = new Engine();
    }

    public Engine getEngine() {
        return engine;
    }
}

class Driver {
    public static void main(String[] args) {
        Car car = new Car();
        car.getEngine().start(); // Violação da Lei de Demeter
    }
}
